# apps/web/app/robots.js. This file governs only the CMS host.
#
# Note this is a crawler instruction, not access control: the API stays publicly
# reachable to anyone who requests it. Rate limiting at the reverse proxy is the
# control for that — see DEPLOYMENT.md.
User-Agent: *
